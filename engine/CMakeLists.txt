cmake_minimum_required(VERSION 3.20)
project(SPENCLMX LANGUAGES CXX)
option(SPENCLMX_BUILD_TESTS "Build unit tests" ON)
add_library(spenclmx_core src/secondary_motion.cpp)
target_include_directories(spenclmx_core PUBLIC include)
target_compile_features(spenclmx_core PUBLIC cxx_std_20)
set_target_properties(spenclmx_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(spenclmx_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(spenclmx_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SPENCLMX_BUILD_TESTS)
  enable_testing()
  add_executable(spenclmx_tests tests/secondary_motion_tests.cpp)
  target_link_libraries(spenclmx_tests PRIVATE spenclmx_core)
  add_test(NAME spenclmx_tests COMMAND spenclmx_tests)
endif()
